这个 demo 验证了什么

  • 首屏不播：刷新直接看到内容，不与首页 hero 自己的入场叠两层。
  • 只淡入内容区：顶部导航是 fixed，切页时既不变暗也不位移。
  • 前进 / 后退不播：浏览器返回时直接切换，滚动位置按历史恢复。
  • 减少动效直接切：勾选面板里的「模拟减少动效」，或系统开启 reduce，动画为 0ms。
  • 整页只动 opacity：文章页顶部那条 fixed 阅读进度条，在正确模式下不会跑位； 打开「上移 6px」的错误示范后，它会跟着内容一起下移 —— 这就是不用 transform 包整页的原因。
真实接入时怎么改（示意代码，不是本站产物） 
// src/app/components/PageMain.tsx（示意，约 40 行）
'use client';
import { useEffect, useLayoutEffect, useRef } from 'react';
import { usePathname } from 'next/navigation';

const SKIP = ['/chat'];
const useIsoLayoutEffect =
  typeof window === 'undefined' ? useEffect : useLayoutEffect;

export default function PageMain({ children }: { children: React.ReactNode }) {
  const pathname = usePathname();
  const ref = useRef<HTMLElement>(null);
  const first = useRef(true);
  const skipPop = useRef(false);

  useEffect(() => {
    const onPop = () => { skipPop.current = true; };
    window.addEventListener('popstate', onPop);
    return () => window.removeEventListener('popstate', onPop);
  }, []);

  useIsoLayoutEffect(() => {
    if (first.current) { first.current = false; return; }
    if (skipPop.current) { skipPop.current = false; return; }
    if (SKIP.some((p) => pathname === p || pathname.startsWith(`${p}/`))) return;
    const el = ref.current;
    if (!el) return;
    el.classList.remove('page-enter');
    void el.offsetWidth; // 强制 reflow，重启动画
    el.classList.add('page-enter');
  }, [pathname]);

  return <main ref={ref}>{children}</main>;
}

/* 样式：只动 opacity，不写 forwards（结束后释放，不残留 stacking context） */
@keyframes page-in { from { opacity: 0; } }
.page-enter { animation: page-in 160ms ease-out backwards; }
@media (prefers-reduced-motion: reduce) {
  .page-enter { animation: none; }
}
